# ---------------------------------------------------------------------------
# MathNodes FIAT Gateway - runtime requirements
#
# Target interpreter: Python 3.8 - 3.13 (Linux server and macOS dev tested).
#
# Install:
#   python3 -m venv .venv
#   . .venv/bin/activate
#   pip install -r requirements.txt
#
# This file lists ONLY what the application (fiat_api.py / uWSGI.py) actually
# imports. Direct dependencies are pinned; the few critical transitive
# dependencies (Werkzeug, SQLAlchemy, PyMySQL) are pinned too because the app
# breaks with the unconstrained latest versions.
# ---------------------------------------------------------------------------

# --- Web framework ---------------------------------------------------------
Flask==2.2.5
Werkzeug==2.3.8           # pin: Flask 2.2.x is not compatible with Werkzeug 3.x
Flask-SQLAlchemy==2.5.1   # pin: 2.5.x requires SQLAlchemy < 2.0 (see below)
Flask-HTTPAuth==4.8.0

# --- Database --------------------------------------------------------------
SQLAlchemy==1.4.54        # last 1.x release; required by Flask-SQLAlchemy 2.5.1
Flask-MySQL==1.5.2        # provides `flaskext.mysql.MySQL`
PyMySQL==1.1.0            # MySQL driver used by Flask-MySQL

# --- Auth / crypto ---------------------------------------------------------
passlib==1.7.4            # pin: 1.8+ removed passlib.apps.custom_app_context
PyJWT==2.8.0

# --- Payments --------------------------------------------------------------
stripe==11.2.0

# --- Process control (drives the `sentinelhub` CLI for payouts) ------------
pexpect==4.9.0

# --- WSGI server -----------------------------------------------------------

#   - Linux  : apt-get install -y uwsgi   (or uncomment + install build deps:
#              build-essential python3-dev libpcre3-dev)
#   - macOS  : brew install uwsgi
